CREATE OR REPLACE WAREHOUSE demo_wh
WITH WAREHOUSE_SIZE = 'XSMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE;

create or replace database s_db;

create or replace schema public;


MERGE INTO curent_customer t
USING (
SELECT customer_id, first_name, last_name, email, signup_date, country
FROM customer_data
) s
ON t.customer_id = s.customer_id
OR (t.first_name = s.first_name AND t.last_name = s.last_name) -- match on name
WHEN MATCHED THEN
UPDATE SET
t.email = s.email,
t.signup_date = s.signup_date,
t.country = s.country
WHEN NOT MATCHED THEN
INSERT (customer_id, first_name, last_name, email, signup_date, country)
